cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED IDF_TARGET AND NOT DEFINED ENV{IDF_TARGET})
    set(IDF_TARGET "esp32c6" CACHE STRING "ESP chip target")
endif()
if(DEFINED ENV{IDF_TARGET} AND NOT DEFINED IDF_TARGET)
    set(IDF_TARGET "$ENV{IDF_TARGET}")
endif()
if(NOT IDF_TARGET MATCHES "^(esp32c6|esp32s3)$")
    message(FATAL_ERROR "Kubik supports esp32c6 or esp32s3")
endif()
set(KUBIK_CHARACTER "TESS" CACHE STRING "Firmware character: TESS or PLUSH")
if(IDF_TARGET STREQUAL "esp32s3")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Selected IDF configuration" FORCE)
    if(KUBIK_CHARACTER STREQUAL "PLUSH")
        set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults.plush")
    else()
        set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults")
    endif()
elseif(KUBIK_CHARACTER STREQUAL "PLUSH")
    set(PLUSH_CONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../tools/configure-plush.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig" "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults" "${PLUSH_CONFIG}"
        COMMAND_ERROR_IS_FATAL ANY)
    set(SDKCONFIG "${PLUSH_CONFIG}" CACHE FILEPATH "Selected IDF configuration" FORCE)
endif()
if(DEFINED SDKCONFIG)
    set(PRINTER_CONFIG "${SDKCONFIG}")
elseif(DEFINED ENV{SDKCONFIG})
    set(PRINTER_CONFIG "$ENV{SDKCONFIG}")
else()
    set(PRINTER_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig")
endif()
execute_process(COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/../tools/configure-printer.py"
    "${PRINTER_CONFIG}" "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults" "${IDF_TARGET}"
    COMMAND_ERROR_IS_FATAL ANY)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(IDF_TARGET STREQUAL "esp32s3")
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.esp32s3")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/main/version.h" VERSION_LINE REGEX "^#define KUBIK_FW_VERSION")
string(REGEX MATCH "\"([^\"]+)\"" VERSION_MATCH "${VERSION_LINE}")
set(FIRMWARE_VERSION "${CMAKE_MATCH_1}")
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git diff --quiet HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    RESULT_VARIABLE SOURCE_DIRTY)
set(PROJECT_VER "${FIRMWARE_VERSION}-${SOURCE_REVISION}")
if(SOURCE_DIRTY)
    string(APPEND PROJECT_VER "-dirty")
endif()
project(kubik)

# `idf.py flash` builds only the flash target's own dependencies: make it repack the assets too.
add_dependencies(flash assets_image)
